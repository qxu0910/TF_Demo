cmake_minimum_required(VERSION 3.16)
project(token_factory_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(runtime src/runtime.cpp src/scheduler.cpp)
target_link_libraries(runtime PUBLIC Threads::Threads)
target_include_directories(runtime PUBLIC include)
target_compile_features(runtime PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(runtime PUBLIC /W4 /utf-8)
else()
    target_compile_options(runtime PUBLIC -Wall -Wextra -Wpedantic)
endif()

add_executable(runtime_demo src/main.cpp)
target_link_libraries(runtime_demo PRIVATE runtime)

enable_testing()
add_executable(runtime_tests tests/runtime_tests.cpp)
target_link_libraries(runtime_tests PRIVATE runtime)
add_test(NAME runtime_tests COMMAND runtime_tests)

add_executable(scheduler_tests tests/scheduler_tests.cpp)
target_link_libraries(scheduler_tests PRIVATE runtime)
add_test(NAME scheduler_tests COMMAND scheduler_tests)

option(BUILD_RUNTIME_SERVER "Build the HTTP runtime (downloads pinned headers)" ON)
set(RUNTIME_DEPS_DIR "${CMAKE_BINARY_DIR}/_deps/headers" CACHE PATH "Directory containing pinned HTTP and JSON headers")
if(BUILD_RUNTIME_SERVER)
    file(MAKE_DIRECTORY "${RUNTIME_DEPS_DIR}")
    function(pinned_header name url sha)
        set(path "${RUNTIME_DEPS_DIR}/${name}")
        if(NOT EXISTS "${path}")
            file(DOWNLOAD "${url}" "${path}" EXPECTED_HASH "SHA256=${sha}" TLS_VERIFY ON TIMEOUT 120)
        endif()
        file(SHA256 "${path}" actual)
        if(NOT actual STREQUAL sha)
            message(FATAL_ERROR "Checksum mismatch: ${name}")
        endif()
    endfunction()
    pinned_header(httplib.h "https://raw.githubusercontent.com/yhirose/cpp-httplib/v0.20.0/httplib.h" "2831f48258d8e20ea88ea82c63f15014ced0a584cbdb1cd3bea1a60b6fb971e5")
    pinned_header(json.hpp "https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp" "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    add_executable(runtime_server src/server.cpp)
    target_include_directories(runtime_server SYSTEM PRIVATE "${RUNTIME_DEPS_DIR}")
    target_link_libraries(runtime_server PRIVATE runtime)
    if(WIN32)
        target_link_libraries(runtime_server PRIVATE ws2_32 crypt32)
    endif()
endif()
